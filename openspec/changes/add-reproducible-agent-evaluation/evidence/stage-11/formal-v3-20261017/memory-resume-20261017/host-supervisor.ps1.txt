using System;
using System.Runtime.InteropServices;
public class NovaMemory {
 [StructLayout(LayoutKind.Sequential)] public struct Status {
  public uint Length,Load; public ulong TotalPhys,AvailPhys,TotalPage,AvailPage,TotalVirtual,AvailVirtual,AvailExtended;
 }
 [DllImport("kernel32.dll",SetLastError=true)] public static extern bool GlobalMemoryStatusEx(ref Status s);
 public static Status Read() { var s=new Status(); s.Length=(uint)Marshal.SizeOf(s); if(!GlobalMemoryStatusEx(ref s)) throw new Exception("memory query failed"); return s; }
}
'@

function Get-Memory {
 $s=[NovaMemory]::Read()
 $vm=0L
 foreach($p in [System.Diagnostics.Process]::GetProcessesByName('vmmemWSL')) { $vm += $p.PrivateMemorySize64; $p.Dispose() }
 return @{commit=[long]$s.AvailPage;physical=[long]$s.AvailPhys;vm=$vm}
}
function Can-Start($m) {
 # 给 WSL 剩余增长和宿主各留余量；已提交的 VM 内存无需重复预留。
 $required=[long](3GB+[Math]::Max([long]0,[long](6.5GB-$m.vm)))
 return $m.commit -ge $required -and $m.physical -ge 1.5GB
}
function Is-Low($m) { return $m.commit -lt 2GB -or $m.physical -lt 0.75GB }
if($SelfCheck) {
 if(!(Can-Start @{commit=8.5GB;physical=2GB;vm=1GB})) {throw 'start boundary'}
 if(Can-Start @{commit=8.5GB-1;physical=2GB;vm=1GB}) {throw 'growth headroom'}
 if(!(Can-Start @{commit=3GB;physical=2GB;vm=6.5GB})) {throw 'already committed VM counted twice'}
 if(!(Is-Low @{commit=2GB-1;physical=2GB;vm=1GB})) {throw 'commit low boundary'}
 if(!(Is-Low @{commit=8GB;physical=0.75GB-1;vm=1GB})) {throw 'physical low boundary'}
 if(Is-Low @{commit=2GB;physical=0.75GB;vm=1GB}) {throw 'low boundary equality'}
 '6 memory policy checks passed; no model calls'
 exit 0
}

function Set-State($phase,$extra=@{}) {
 $m=Get-Memory
 $state=[ordered]@{phase=$phase;host_pid=$PID;updated_utc=[DateTime]::UtcNow.ToString('o');recorded=$Recorded;sequence=$Sequence;planned=216;linux_minimum_new_run_bytes=1073741824;host_memory=$m;state_path=(Join-Path $HostRoot 'status.json');campaign_id='novacode-compact-20261017-v3-r1';manifest_sha256='0cf2d3fb6ba40d1f4f62b17d31cdb975c19eb5b2adb0ac89d0572922f703fa2f'}

}
function Start-Adapter($mode,$row) {
 $script:Counter++
 $tag='{0:D4}-{1}-{2}' -f $script:Counter,$mode,$row
 $out=Join-Path $HostRoot ($tag+'.stdout.log')
 $err=Join-Path $HostRoot ($tag+'.stderr.log')
 $launchArgs=@('-d','Ubuntu-22.04','-u','louishyh','--cd','/home/louishyh/projects/NovaCode','--','/usr/bin/env','PYTHONPATH=src:.','PATH=/snap/bin:/usr/bin:/bin:/home/louishyh/.local/bin:/home/louishyh/projects/NovaCode/.venv/bin','/home/louishyh/projects/NovaCode/.venv/bin/python',$Adapter,$mode,[string]$row)
 $p=Start-Process -FilePath 'C:\Windows\System32\wsl.exe' -ArgumentList $launchArgs -WindowStyle Hidden -PassThru -RedirectStandardOutput $out -RedirectStandardError $err
 return @{process=$p;stdout=$out;stderr=$err}
}
function Invoke-Adapter($mode,$row) {
 $job=Start-Adapter $mode $row
 $job.process.WaitForExit()
 $code=$job.process.ExitCode
 if($code -ne 0) {throw "WSL adapter $mode failed ($code); logs: $($job.stderr)"}
 return @(Get-Content -LiteralPath $job.stdout -Encoding UTF8|Where-Object {$_ -match '^\{'})[-1]|ConvertFrom-Json
}
function Wait-For-Memory {
 $since=$null
 while($true) {
  $m=Get-Memory
  $c=[System.IO.DriveInfo]::new('C:\').AvailableFreeSpace
  if((Can-Start $m) -and $c -ge 5GB) {
   if($null -eq $since) {$since=[DateTime]::UtcNow}
   if(([DateTime]::UtcNow-$since).TotalSeconds -ge 30) {return}
  } else {$since=$null}
  Set-State 'PAUSED_MEMORY' @{reason='等待宿主余量恢复并稳定 30 秒；不调用 WSL、不派发新模型任务';c_free_bytes=$c}
  Start-Sleep -Seconds 10
 }
}
