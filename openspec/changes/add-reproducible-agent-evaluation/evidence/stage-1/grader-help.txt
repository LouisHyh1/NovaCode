Run evaluation harness for the given dataset and predictions.

options:
  -h, --help            show this help message and exit
  --dataset_name DATASET_NAME
                        Name of dataset or path to JSON file. (default: SWE-
                        bench/SWE-bench_Lite)
  --split SPLIT         Split of the dataset (default: test)
  --instance_ids INSTANCE_IDS [INSTANCE_IDS ...]
                        Instance IDs to run (space separated) (default: None)
  --predictions_path PREDICTIONS_PATH
                        Path to predictions file - if 'gold', uses gold
                        predictions (default: None)
  --max_workers MAX_WORKERS
                        Maximum number of workers (should be <= 75% of CPU
                        cores) (default: 4)
  --open_file_limit OPEN_FILE_LIMIT
                        Open file limit (default: 4096)
  --timeout TIMEOUT     Timeout (in seconds) for running tests for each
                        instance (default: 1800)
  --force_rebuild FORCE_REBUILD
                        Force rebuild of all images (default: False)
  --cache_level {none,base,env,instance}
                        Cache level - remove images above this level (default:
                        env)
  --clean CLEAN         Clean images above cache level (default: False)
  --run_id RUN_ID       Run ID - identifies the run (default: None)
  --namespace NAMESPACE
                        Namespace for images (default: swebench)
  --instance_image_tag INSTANCE_IMAGE_TAG
                        Instance image tag (default: latest)
  --rewrite_reports REWRITE_REPORTS
                        Doesn't run new instances, only writes reports for
                        instances with existing test outputs (default: False)
  --report_dir REPORT_DIR
                        Directory to write reports to (default: .)
  --modal MODAL         Run on Modal (default: False)
